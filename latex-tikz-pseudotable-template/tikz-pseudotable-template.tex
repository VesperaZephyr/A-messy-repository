\def\tcfgcolwidths{}
\def\tcfgtablew{\textwidth}
\def\tcfgheaderrow{1}
\def\tcfggridstyle{all}
\def\tcfgminrow{0.72cm}

\newcounter{gsumc}
% ---- 设置命令（API：每个表调用一次；取值存进内部 \tcfg* 宏）----
\newcommand{\tcols}[1]{\def\tcfgcols{#1}}
\newcommand{\trows}[1]{\def\tcfgrows{#1}}
\newcommand{\tcolwidths}[1]{\def\tcfgcolwidths{#1}}
\newcommand{\ttablew}[1]{\def\tcfgtablew{#1}}
\newcommand{\theaderrow}[1]{\def\tcfgheaderrow{#1}}
\newcommand{\tgridstyle}[1]{\def\tcfggridstyle{#1}}
\newcommand{\tminrow}[1]{\def\tcfgminrow{#1}}

% ---- 颜色 ----
\definecolor{tblhead}{RGB}{217,236,247}   % 表头淡蓝
\definecolor{tblrow}{RGB}{245,248,252}    % 隔行淡色
\definecolor{tblhi}{RGB}{255,235,205}     % 高亮（暖橙）

% ---- 每列对齐（可选）：\tcolalign{列号}{left|center|right} ----
\newcommand{\tcolalign}[2]{\expandafter\def\csname tca#1\endcsname{#2}}
\newcommand{\tcaget}[1]{\ifcsname tca#1\endcsname\csname tca#1\endcsname\else center\fi}

% ---- 读配置：算列宽 + 生成 matrix 选项 ----
\newcommand{\preparetable}{%
  % 列宽比例之和（\xdef 全局累加，避免 \foreach 分组丢值；
  %  均分时每列比例为 1，和=列数）
  \ifx\tcfgcolwidths\empty
    \edef\cwsum{\tcfgcols}%
  \else
    \xdef\cwsum{0}%
    \foreach \w in \tcfgcolwidths {%
      \pgfmathsetmacro{\cwsumnew}{\cwsum+\w}%
      \xdef\cwsum{\cwsumnew}%
    }%
  \fi
  % 列宽（pt）
  \ifx\tcfgcolwidths\empty
    \foreach \i in {1,...,\tcfgcols} {%
      \pgfmathparse{1/\cwsum*\the\tcfgtablew}%
      \expandafter\xdef\csname cw\i\endcsname{\pgfmathresult}%
    }%
  \else
    \foreach[count=\i] \w in \tcfgcolwidths {%
      \pgfmathparse{\w/\cwsum*\the\tcfgtablew}%
      \expandafter\xdef\csname cw\i\endcsname{\pgfmathresult}%
    }%
  \fi
  % 全网格时给单元格加框，否则不加
  \ifdefstring{\tcfggridstyle}{all}{\def\griddraw{, draw, line width=0.5pt}}{\def\griddraw{}}%
  % 构建 matrix 选项（用 \xdef 把循环变量立即展开，避免存成字面量）
  \gdef\tblmatrixopts{%
    matrix of nodes, nodes in empty cells, inner sep=0pt, outer sep=0pt,
    column sep=0pt, row sep=0pt,
    nodes={align=center, inner sep=4pt, font=\small, minimum height=\tcfgminrow\griddraw}%
  }%
  \foreach \i in {1,...,\tcfgcols} {%
    \xdef\tblmatrixopts{\unexpanded\expandafter{\tblmatrixopts}, column \i/.style={text width=\csname cw\i\endcsname pt, align=\tcaget{\i}}}%
  }%
  \ifnum\tcfgheaderrow>0
    \xdef\tblmatrixopts{\unexpanded\expandafter{\tblmatrixopts}, row \tcfgheaderrow/.style={nodes={fill=tblhead}}}%
  \fi
  \ifzebra
    \foreach \r in {1,...,\tcfgrows} {%
      \ifnum\r>\tcfgheaderrow
        \ifodd\r\relax\else
          \xdef\tblmatrixopts{\unexpanded\expandafter{\tblmatrixopts}, row \r/.style={nodes={fill=tblrow}}}%
        \fi
      \fi
    }%
  \fi
  % 注册成 style。必须用 \xdef 把 \tblmatrixopts 彻底展开成字面 token 序列再定义
  % style——否则 style 里存的是宏 token，pgfkeys 不会把宏展开结果按逗号拆成多个
  % key，会报 "I do not know the key '/tikz/matrix of nodes, ...'"。
  \xdef\tblstyle{\noexpand\tikzset{tblstyle/.style={\tblmatrixopts}}}\tblstyle%
}

% ---- 合并单元格：\mergeblock[样式]{起始行}{起始列}{跨行}{跨列}{文字} ----
% 样式缺省 fill=white,draw=black（白色覆盖内部网格线，再描合并区外框）。
% 若合并落在表头，需传 fill=tblhead 以保持表头颜色。
\newcommand{\mergeblock}[6][fill=white, draw=black, line width=0.5pt]{%
  \pgfmathtruncatemacro{\rB}{#2+#4-1}%
  \pgfmathtruncatemacro{\cB}{#3+#5-1}%
  \node[fit=(m-#2-#3)(m-\rB-\cB), inner sep=0pt, #1, align=center, font=\small] {#6};
}

% ---- 收尾：按 \tgridstyle 画外框 / 横线 / 竖线 ----
\newcommand{\finishtable}{%
  \ifdefstring{\tcfggridstyle}{all}{%
    \draw[line width=1pt] (m-1-1.north west) rectangle (m-\tcfgrows-\tcfgcols.south east);
  }{}%
  \ifdefstring{\tcfggridstyle}{outer}{%
    \draw[line width=1pt] (m-1-1.north west) rectangle (m-\tcfgrows-\tcfgcols.south east);
  }{}%
  \ifdefstring{\tcfggridstyle}{hlines}{%
    \draw[line width=1pt] (m-1-1.north west) -- (m-1-\tcfgcols.north east);
    \draw[line width=1pt] (m-\tcfgrows-1.south west) -- (m-\tcfgrows-\tcfgcols.south east);
    \foreach \r in {1,...,\tcfgrows} {%
      \ifnum\r<\tcfgrows
        \draw[line width=0.5pt] (m-\r-1.south west) -- (m-\r-\tcfgcols.south east);
      \fi
    }%
  }{}%
  \ifdefstring{\tcfggridstyle}{vlines}{%
    \draw[line width=1pt] (m-1-1.north west) -- (m-\tcfgrows-1.south west);
    \draw[line width=1pt] (m-1-\tcfgcols.north east) -- (m-\tcfgrows-\tcfgcols.south east);
    \foreach \c in {1,...,\tcfgcols} {%
      \ifnum\c<\tcfgcols
        \draw[line width=0.5pt] (m-1-\c.north east) -- (m-\tcfgrows-\c.south east);
      \fi
    }%
  }{}%
}

% ============================================================
%  方法三：纯手绘规则网格 helper（用于完全自定义/不对称边框）
% ============================================================
% \drawgridspec{列宽比例(逗号分隔)}{行高cm(逗号分隔)}
% 自动画外框 + 全网格；如需断线，在它之后手动 \draw 局部线段覆盖即可。
\newcommand{\drawgridspec}[2]{%
  % 列宽比例之和（\xdef 全局累加）
  \xdef\gsum{0}%
  \foreach \w in {#1}{%
    \pgfmathsetmacro{\gsumnew}{\gsum+\w}%
    \xdef\gsum{\gsumnew}%
  }%
  \xdef\gxone{0}\pgfmathsetmacro{\gxacc}{0}%
  \foreach[count=\gi] \w in {#1}{%
    \pgfmathparse{\gxacc+\w/\gsum*\the\tcfgtablew}%
    \expandafter\xdef\csname gx\the\numexpr\gi+1\endcsname{\pgfmathresult}%
    \xdef\gxacc{\pgfmathresult}%
    \xdef\gcols{\gi}%
  }%
  \xdef\gyone{0}\pgfmathsetmacro{\gyacc}{0}%
  \foreach[count=\gj] \h in {#2}{%
    \pgfmathparse{\gyacc-\h*28.4527}%  % 行高按 cm 传入，此处换算成 pt
    \expandafter\xdef\csname gy\the\numexpr\gj+1\endcsname{\pgfmathresult}%
    \xdef\gyacc{\pgfmathresult}%
    \xdef\grows{\gj}%
  }%
  \draw[line width=1pt] (\gxone pt,\gyone pt)
        rectangle (\csname gx\the\numexpr\gcols+1\endcsname pt,\csname gy\the\numexpr\grows+1\endcsname pt);
  \foreach \k in {2,...,\gcols}{%
    \draw[line width=0.5pt] (\csname gx\k\endcsname pt,\gyone pt)
          -- (\csname gx\k\endcsname pt,\csname gy\the\numexpr\grows+1\endcsname pt);
  }%
  \foreach \l in {2,...,\grows}{%
    \draw[line width=0.5pt] (\gxone pt,\csname gy\l\endcsname pt)
          -- (\csname gx\the\numexpr\gcols+1\endcsname pt,\csname gy\l\endcsname pt);
  }%
}

% ====================== 使用要点 / API 速查 ======================
\noindent\small\textbf{使用要点（方法二通用引擎）：}
\begin{enumerate}[label=\arabic*.]
  \item 每个表先配置（按需调用，最后 \verb!\preparetable!）：
        列数 \verb!\tcols{4}!、行数 \verb!\trows{5}!、列宽比例 \verb!\tcolwidths{0.16,0.30,0.20,0.34}!
        （留空则均分）、表头行 \verb!\theaderrow{1}!（0=无表头）、斑马纹
        \verb!\zebratrue! / \verb!\zebrafalse!、边框风格 \verb!\tgridstyle{all}!
        （可选 all\,/\,outer\,/\,hlines\,/\,vlines）、逐列对齐 \verb!\tcolalign{1}{left}!。
  \item 写表：\verb!\matrix (m)[tblstyle] { 第一行&第二列&... \\ 第二行 \\ ... };!。
        单元格文字支持中文、逗号、公式；单元格高亮在文字前写 \verb!|[fill=tblhi]|!。
  \item 合并单元格：\verb!\mergeblock[样式]{起始行}{起始列}{跨行}{跨列}{文字}!
        （跨行、跨列、矩形区块皆可；被合并覆盖的格子在 matrix 里留空），
        之后 \verb!\finishtable! 收尾画外框/横线/竖线。
  \item 四种边框：\texttt{all}=全网格；\texttt{outer}=仅外框；\texttt{hlines}=仅横线+上下框；
        \texttt{vlines}=仅竖线+左右框。覆盖常见 Word 边框组合。
  \item 方法三纯手绘：\verb!\drawgridspec{列宽比例}{行高cm}! 自动画规则网格，
        之后手动 \verb!\draw! 局部线段即可做不对称/断线边框，适合极特殊版式。
\end{enumerate}
